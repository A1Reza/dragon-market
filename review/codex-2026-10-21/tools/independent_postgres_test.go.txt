func TestReviewerOracleFallbackAndLockedPrice(t *testing.T) {
    _,_,orders,_,_,_:=researchFixture(t)
    listing,err:=orders.List(context.Background(),"research-item","seller",100);if err!=nil {t.Fatal(err)}
    mode:="valid"
    server:=httptest.NewServer(http.HandlerFunc(func(w http.ResponseWriter,r *http.Request){
        switch mode {
        case "zero":w.Write([]byte(`{"base_price":0}`))
        case "negative":w.Write([]byte(`{"base_price":-3}`))
        case "malformed":w.Write([]byte(`{oops`))
        case "http503":w.WriteHeader(503)
        case "timeout":<-r.Context().Done()
        default:w.Write([]byte(`{"base_price":250}`))
        }
    }));defer server.Close()
    client:=server.Client();client.Timeout=50*time.Millisecond
    upstream,err:=oracle.NewHTTPOracle(server.URL,client);if err!=nil {t.Fatal(err)}
    resilient:=oracle.NewResilientOracle(upstream)
    for _,scenario:=range []string{"valid","zero","negative","malformed","http503","timeout"} {
        mode=scenario
        value,err:=resilient.GetBasePrice(context.Background(),"research-item")
        if err!=nil||value!=250 {t.Fatalf("%s fallback=%d err=%v",scenario,value,err)}
        if err:=NewItemRepository(testDB).UpdateBasePrice(context.Background(),"research-item",value);err!=nil {t.Fatal(err)}
        var price,orderPrice int
        if err:=testDB.QueryRow(`SELECT base_price,(SELECT price FROM limit_orders WHERE id=$1) FROM items WHERE id='research-item'`,listing.ID).Scan(&price,&orderPrice);err!=nil {t.Fatal(err)}
        if price!=250||orderPrice!=100 {t.Fatalf("%s base/listing=%d/%d",scenario,price,orderPrice)}
        t.Logf("scenario=%s fallback=%d SQL base_price=%d listing_price=%d",scenario,value,price,orderPrice)
    }
}

func TestReviewerDownMigrationRefusesAndPreserves(t *testing.T) {
    w, _, _, _, _, _ := researchFixture(t)
    if err := w.Spend(context.Background(), "buyer", 123); err != nil { t.Fatal(err) }
    before := persistedResearchState(t)
    b, err := os.ReadFile("../../migrations/down/000011_daily_accounting_v2.down.sql")
    if err != nil { t.Fatal(err) }
    if _, err = testDB.Exec(string(b)); err == nil || !strings.Contains(err.Error(), "cannot automatically downgrade") {
        t.Fatalf("expected populated downgrade refusal, got %v", err)
    }
    var version, gold, spent int
    if err = testDB.QueryRow(`SELECT daily_accounting_version,gold,daily_spent FROM guilds WHERE id='buyer'`).Scan(&version,&gold,&spent); err != nil { t.Fatal(err) }
    if version != 2 || gold != 9877 || spent != 123 { t.Fatalf("data changed: %d/%d/%d", version,gold,spent) }
    t.Logf("refused down migration; version=%d gold=%d spent=%d; before=%+v after=%+v",version,gold,spent,before,persistedResearchState(t))
}

type reviewerBarrierWallet struct {
    wallet *guild.WalletServiceImpl
    arrived *sync.WaitGroup
}
func(w reviewerBarrierWallet) Spend(ctx context.Context,id string,amount gold.Amount)error {
    if err:=w.wallet.Spend(ctx,id,amount);err!=nil {return err}
    // Force an ordinary possible interleaving: both buyer rows locked before seller credit.
    w.arrived.Done()
    done:=make(chan struct{});go func(){w.arrived.Wait();close(done)}()
    select {case <-done:return nil;case <-ctx.Done():return ctx.Err()}
}
func(w reviewerBarrierWallet) Earn(ctx context.Context,id string,amount gold.Amount)error {
    return w.wallet.Earn(ctx,id,amount)
}

func TestReviewerReciprocalPurchasesDoNotDeadlock(t *testing.T) {
    wallet,items,_,_,_,tx:=researchFixture(t)
    _,err:=testDB.Exec(`UPDATE guilds SET daily_limit=0 WHERE id IN ('buyer','seller');
      INSERT INTO items(id,name,type,owner_id,status,base_price) VALUES('reviewer-reverse','Reverse','rare','buyer','free',100)`)
    if err!=nil {t.Fatal(err)}
    normal:=order.NewOrderService(NewOrderRepository(testDB),wallet,items,tx)
    a,err:=normal.List(context.Background(),"research-item","seller",100);if err!=nil {t.Fatal(err)}
    b,err:=normal.List(context.Background(),"reviewer-reverse","buyer",100);if err!=nil {t.Fatal(err)}
    barrier:=&sync.WaitGroup{};barrier.Add(2)
    svc:=order.NewOrderService(NewOrderRepository(testDB),reviewerBarrierWallet{wallet,barrier},items,tx)
    ctx,cancel:=context.WithTimeout(context.Background(),15*time.Second);defer cancel()
    results:=make(chan error,2)
    go func(){results<-svc.Buy(ctx,a.ID,"buyer")}()
    go func(){results<-svc.Buy(ctx,b.ID,"seller")}()
    x,y:=<-results,<-results
    var total,audits,sold int
    if err:=testDB.QueryRow(`SELECT (SELECT sum(gold) FROM guilds),(SELECT count(*) FROM wallet_transactions),(SELECT count(*) FROM limit_orders WHERE status='sold')`).Scan(&total,&audits,&sold);err!=nil {t.Fatal(err)}
    t.Logf("concurrent reciprocal results=%v / %v; SQL gold total=%d audit rows=%d sold=%d; state=%+v",x,y,total,audits,sold,persistedResearchState(t))
    if total!=30000 {t.Fatalf("MONEY CONSERVATION FAILURE total=%d",total)}
    if x!=nil||y!=nil {t.Fatalf("valid reciprocal purchases cannot both complete: %v / %v",x,y)}
}
